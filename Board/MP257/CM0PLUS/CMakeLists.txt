set(STM32CUBE_MP2_ROOT "$ENV{STM32CUBE_MP2_ROOT}" CACHE PATH "STM32CubeMP2 V1.3.0 installation")
if(NOT STM32CUBE_MP2_ROOT)
    set(STM32CUBE_MP2_ROOT "$ENV{HOME}/STM32Cube/Repository/STM32Cube_FW_MP2_V1.3.0")
endif()
file(STRINGS "${STM32CUBE_MP2_ROOT}/package.xml" release REGEX "PackDescription Release=")
if(NOT release MATCHES "Release=\"FW[.]MP2[.]1[.]3[.]0\"")
    message(FATAL_ERROR "This port requires STM32CubeMP2 V1.3.0")
endif()
set(cmsis "${STM32CUBE_MP2_ROOT}/Drivers/CMSIS")
set(device "${cmsis}/Device/ST/STM32MP2xx")
get_filename_component(sdk "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(target sirinos_mp257_m0plus)
add_library(${target} OBJECT receiver.c ../IPC/peer.c
    "${device}/Source/Templates/gcc/startup_stm32mp2xx_m0plus.s"
    "${device}/Source/Templates/system_stm32mp2xx_m0plus.c"
    "${sdk}/Source/Services/RemoteControl/WBus/wbus_framer.c")
target_compile_definitions(${target} PRIVATE CORE_CM0PLUS STM32MP257Dxx)
target_include_directories(${target} PRIVATE "${cmsis}/Include" "${device}/Include")
target_link_libraries(${target} PRIVATE sirinos_config)
set(script "${CMAKE_CURRENT_SOURCE_DIR}/remoteproc.ld")
add_library(sirinos_platform INTERFACE)
add_library(SirinEmbedOS::Platform ALIAS sirinos_platform)
target_link_libraries(sirinos_platform INTERFACE sirinos_config $<TARGET_OBJECTS:sirinos_mp257_m0plus>)
target_link_options(sirinos_platform INTERFACE "LINKER:-T,${script}")
set_property(TARGET sirinos_platform PROPERTY INTERFACE_LINK_DEPENDS "${script}")
include("${sdk}/CMake/Firmware.cmake")
if(SIRINEMBEDOS_BUILD_EXAMPLE)
    sirinos_add_firmware(SirinEmbedOS-MP257-M0Plus main.c)
endif()
