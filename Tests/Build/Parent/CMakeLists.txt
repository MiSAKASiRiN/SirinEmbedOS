cmake_minimum_required(VERSION 3.22)
project(ParentRobot LANGUAGES C CXX ASM)
set(SIRINEMBEDOS_CONFIG_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/robot_config.h" CACHE FILEPATH "")
add_subdirectory("${SIRINEMBEDOS_SOURCE_DIR}" sdk)
if(NOT CMAKE_PROJECT_NAME STREQUAL "ParentRobot" OR NOT PROJECT_NAME STREQUAL "ParentRobot")
 message(FATAL_ERROR "SDK changed parent project")
endif()
if(TARGET SirinEmbedOS)
 message(FATAL_ERROR "SDK unexpectedly created its example in a parent build")
endif()
sirinos_add_firmware(RobotFirmware app.cpp)
add_executable(ManualFirmware app.cpp)
target_link_libraries(ManualFirmware PRIVATE SirinEmbedOS::Platform)

sirinos_add_firmware(PrototypeBindings "${SIRINEMBEDOS_SOURCE_DIR}/Examples/Minimal/app_main.c"
    "${SIRINEMBEDOS_SOURCE_DIR}/Examples/PrototypeBindings/bindings.c")
sirinos_add_firmware(ParameterExample "${SIRINEMBEDOS_SOURCE_DIR}/Examples/Parameters/app_main.cpp")
sirinos_add_firmware(BringUpExample "${SIRINEMBEDOS_SOURCE_DIR}/Examples/BringUp/app_main.c")
sirinos_add_firmware(SensorBringUpExample "${SIRINEMBEDOS_SOURCE_DIR}/Examples/SensorBringUp/app_main.c")
sirinos_add_firmware(INSBringUpExample "${SIRINEMBEDOS_SOURCE_DIR}/Examples/INSBringUp/app_main.c")
sirinos_add_firmware(RawBinding app.cpp raw_binding.cpp)
sirinos_add_firmware(AllComponents app.cpp api_surface.cpp)
target_link_options(AllComponents PRIVATE "LINKER:--no-gc-sections")
