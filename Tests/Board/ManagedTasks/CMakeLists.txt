cmake_minimum_required(VERSION 3.22)
project(ManagedTasksBoard LANGUAGES C CXX ASM)
option(BOARD_STATS "Enable cycle statistics" ON)
option(BOARD_MONITOR "Enable SystemMonitor" ON)
add_compile_options(-g3)
get_filename_component(SDK "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${SDK}" sdk)
sirinos_add_firmware(ManagedTasksBoard app.cpp)
add_library(managed_board_adapter OBJECT kernel_adapter.c adc_fault_adapter.c)
target_link_libraries(managed_board_adapter PRIVATE sirinos_build_internal)
target_include_directories(managed_board_adapter PRIVATE
    "${SDK}/Source/Diagnostics/Runtime" "${SDK}/Board/Support")
target_sources(ManagedTasksBoard PRIVATE $<TARGET_OBJECTS:managed_board_adapter>)
target_link_options(ManagedTasksBoard PRIVATE -Wl,--wrap=SRNOS_BoardADCTake -Wl,--wrap=SRNOS_BoardADCStop)
target_compile_definitions(ManagedTasksBoard PRIVATE BOARD_STATS=$<BOOL:${BOARD_STATS}> BOARD_MONITOR=$<BOOL:${BOARD_MONITOR}>)
