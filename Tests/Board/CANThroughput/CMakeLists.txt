cmake_minimum_required(VERSION 3.22)
project(CANThroughput LANGUAGES C CXX ASM)
add_compile_options(-g3)
get_filename_component(SDK "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${SDK}" sdk)
sirinos_add_firmware(CANThroughput app.cpp)
add_library(can_test_adapter OBJECT adapter.c)
target_link_libraries(can_test_adapter PRIVATE sirinos_build_internal)
target_include_directories(can_test_adapter PRIVATE "${SDK}/Board/Support")
target_sources(CANThroughput PRIVATE $<TARGET_OBJECTS:can_test_adapter>)
