cmake_minimum_required(VERSION 3.22)
project(CAN4SPILoad LANGUAGES C CXX ASM)
get_filename_component(SDK "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${SDK}" sdk)
add_library(can4_load_adapter OBJECT adapter.cpp ../CANFDLoad/adapter.c)
target_link_libraries(can4_load_adapter PRIVATE sirinos_build_internal)
target_include_directories(can4_load_adapter PRIVATE "${SDK}/Board/Support")
function(can_profile name fd rate brs)
    sirinos_add_firmware(${name} app.cpp)
    target_compile_options(${name} PRIVATE -g3)
    target_compile_definitions(${name} PRIVATE SRNOS_TEST_LONG=$<IF:$<STREQUAL:${name},Classic>,1,0> SRNOS_TEST_FD=${fd} SRNOS_TEST_DATA_RATE=${rate} SRNOS_TEST_BRS=${brs})
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:can4_load_adapter>)
endfunction()
can_profile(Classic 0 0 0)
can_profile(FD2M 1 2000000 1)
can_profile(FD5M 1 5000000 1)
can_profile(FD8M 1 8000000 1)
can_profile(FD8MProbe 1 8000000 1)
target_compile_definitions(FD8MProbe PRIVATE SRNOS_TEST_PROBE=1)
can_profile(ClassicQuick 0 0 0)
can_profile(FD4M 1 4000000 1)
can_profile(FD1M 1 1000000 0)

option(SRNOS_TEST_DETAIL_TIMING "Short diagnostic build only; disable for acceptance" OFF)
if(SRNOS_TEST_DETAIL_TIMING)
    target_compile_definitions(sirinos_components PRIVATE SRNOS_CAN4_PROFILE=1)
    target_compile_definitions(ClassicQuick PRIVATE SRNOS_TEST_PROFILE=1)
endif()
