cmake_minimum_required(VERSION 3.22)
project(CANFDLoad LANGUAGES C CXX ASM)
get_filename_component(SDK "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${SDK}" sdk)
add_library(can_fd_load_adapter OBJECT adapter.c)
target_link_libraries(can_fd_load_adapter PRIVATE sirinos_build_internal)
target_include_directories(can_fd_load_adapter PRIVATE "${SDK}/Board/Support")
function(can_profile name fd rate brs)
    sirinos_add_firmware(${name} app.cpp)
    target_compile_options(${name} PRIVATE -g3)
    target_compile_definitions(${name} PRIVATE SRNOS_TEST_FD=${fd} SRNOS_TEST_DATA_RATE=${rate} SRNOS_TEST_BRS=${brs})
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:can_fd_load_adapter>)
endfunction()
can_profile(Classic 0 0 0)
can_profile(FD1M 1 1000000 0)
can_profile(FD2M 1 2000000 1)
can_profile(FD4M 1 4000000 1)
can_profile(FD5M 1 5000000 1)
can_profile(FD8M 1 8000000 1)
